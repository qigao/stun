class AutoSaveManager {
public:
  AutoSaveManager(ModernWhiteboardApp *app)
      : m_app(app), m_has_unsaved_changes(false), m_is_saving(false), m_auto_save_enabled(false) {
    m_last_change_time = std::chrono::steady_clock::now();
  }

  void start() {
    m_auto_save_enabled = true;
    m_last_change_time = std::chrono::steady_clock::now();
  }

  void stop() { m_auto_save_enabled = false; }

  void mark_changed() {
    m_has_unsaved_changes = true;
    m_last_change_time = std::chrono::steady_clock::now();
  }

  void update() {
    if (!m_auto_save_enabled || !m_has_unsaved_changes || m_is_saving) {
      return;
    }

    auto now = std::chrono::steady_clock::now();
    auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(now - m_last_change_time);

    // Auto-save after 30 seconds of inactivity
    if (elapsed.count() >= 30) {
      save_now();
    }
  }

  void save_now();
  bool has_auto_save_data();
  void load_from_local_storage();
  void clear_auto_save();

  bool is_saving() const { return m_is_saving; }

private:
  ModernWhiteboardApp *m_app;
  std::chrono::steady_clock::time_point m_last_change_time;
  bool m_has_unsaved_changes;
  bool m_is_saving;
  bool m_auto_save_enabled;

  void save_to_local_storage();
  std::string get_auto_save_path();
};
