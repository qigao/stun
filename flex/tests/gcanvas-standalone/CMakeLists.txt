cmake_minimum_required(VERSION 3.20...3.31)
project(StunGCanvasRendererQualification LANGUAGES C CXX)
include(CTest)
get_filename_component(STUN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Salts CONFIG REQUIRED PATHS "$ENV{SALTS_ROOT}" NO_DEFAULT_PATH)
foreach(option IN ITEMS OPENGL GLES ANDROID_EGL VULKAN WINDOW TESTS GPU_TESTS EXAMPLES)
  set(GCANVAS_BUILD_${option} OFF CACHE BOOL "Headless renderer qualification" FORCE)
endforeach()
set(GCANVAS_BUILD_SVG ON CACHE BOOL "Use the real production SVG adapter" FORCE)
add_subdirectory("${STUN_ROOT}/shared/gcanvas" "${CMAKE_CURRENT_BINARY_DIR}/gcanvas")
add_executable(test_gcanvas_renderer
  "${STUN_ROOT}/flex/tests/test_gcanvas_renderer.cpp"
  "${STUN_ROOT}/flex/render/engines/gcanvas/renderer.cpp")
target_compile_features(test_gcanvas_renderer PRIVATE cxx_std_17)
target_include_directories(test_gcanvas_renderer PRIVATE "${STUN_ROOT}/flex/include"
  "${STUN_ROOT}/flex/render/engines/gcanvas/include")
target_link_libraries(test_gcanvas_renderer PRIVATE Salts::TinyTest gCanvas::Core gCanvas::Svg)
foreach(target IN ITEMS test_gcanvas_renderer gcanvas_core gcanvas_svg)
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
endforeach()
add_test(NAME test_gcanvas_renderer COMMAND test_gcanvas_renderer)
set_tests_properties(test_gcanvas_renderer PROPERTIES TIMEOUT 30)
