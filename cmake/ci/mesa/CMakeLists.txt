cmake_minimum_required(VERSION 3.20)
project(mesa_lifecycle_control LANGUAGES CXX)
find_package(glfw3 CONFIG REQUIRED)
find_package(glad CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(mesa_driver_control driver_control.cpp)
target_link_libraries(mesa_driver_control PRIVATE glfw glad::glad Threads::Threads)
target_compile_features(mesa_driver_control PRIVATE cxx_std_17)
target_compile_options(mesa_driver_control PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined)
target_link_options(mesa_driver_control PRIVATE -fsanitize=address,undefined)
