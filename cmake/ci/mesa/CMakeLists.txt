cmake_minimum_required(VERSION 3.20)
project(mesa_lifecycle_control LANGUAGES C CXX)

if(NOT STUN_MESA_GLFW_SOURCE OR NOT EXISTS "${STUN_MESA_GLFW_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Mesa lifecycle control requires STUN_MESA_GLFW_SOURCE")
endif()
if(NOT STUN_MESA_GLAD_SOURCE OR NOT EXISTS "${STUN_MESA_GLAD_SOURCE}/cmake/CMakeLists.txt")
  message(FATAL_ERROR "Mesa lifecycle control requires STUN_MESA_GLAD_SOURCE")
endif()

set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_X11 ON CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
add_subdirectory("${STUN_MESA_GLFW_SOURCE}" "${CMAKE_BINARY_DIR}/glfw" EXCLUDE_FROM_ALL)

set(GLAD_SOURCES_DIR "${STUN_MESA_GLAD_SOURCE}" CACHE STRING "" FORCE)
add_subdirectory("${STUN_MESA_GLAD_SOURCE}/cmake" "${CMAKE_BINARY_DIR}/glad" EXCLUDE_FROM_ALL)
glad_add_library(stun_mesa_glad REPRODUCIBLE API gl:core=4.1)

find_package(Threads REQUIRED)
add_executable(mesa_driver_control driver_control.cpp)
target_link_libraries(mesa_driver_control PRIVATE glfw stun_mesa_glad Threads::Threads)
target_compile_features(mesa_driver_control PRIVATE cxx_std_17)
target_compile_options(mesa_driver_control PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined)
target_link_options(mesa_driver_control PRIVATE -fsanitize=address,undefined)
